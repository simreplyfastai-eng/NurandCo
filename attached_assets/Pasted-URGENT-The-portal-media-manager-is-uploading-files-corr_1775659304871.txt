FIX 2 — index.html on page load:
Add this fetch call that runs 
immediately when the page loads:

fetch('/api/media/config')
  .then(r => r.json())
  .then(config => {
    
    // Before & After images
    if (config.beforeAfter) {
      config.beforeAfter.forEach((item, i) => {
        if (item.src) {
          const img = document.querySelector(
            '[data-ba-slot="' + (i+1) + '"] img'
          )
          if (img) img.src = item.src
          const label = document.querySelector(
            '[data-ba-slot="' + (i+1) + '"] .ba-label'
          )
          if (label && item.label) 
            label.textContent = item.label
        }
      })
    }
    
    // Results videos
    if (config.videos) {
      config.videos.forEach((item, i) => {
        if (item.src) {
          const video = document.querySelector(
            '[data-video-slot="' + (i+1) + '"] video'
          )
          if (video) {
            video.src = item.src
            video.load()
          }
          const label = document.querySelector(
            '[data-video-slot="' + (i+1) + '"] .video-label'
          )
          if (label && item.label) 
            label.textContent = item.label
        }
      })
    }
    
    // Practitioner image
    if (config.practitionerImage) {
      const img = document.getElementById(
        'practitioner-photo'
      )
      if (img) img.src = config.practitionerImage
    }
    
    // Graduate photos
    if (config.graduates) {
      config.graduates.forEach((item, i) => {
        if (item.src) {
          const img = document.querySelector(
            '[data-grad-slot="' + (i+1) + '"] img'
          )
          if (img) img.src = item.src
          const name = document.querySelector(
            '[data-grad-slot="' + (i+1) + '"] .grad-name'
          )
          if (name && item.name) 
            name.textContent = item.name
          const course = document.querySelector(
            '[data-grad-slot="' + (i+1) + '"] .grad-course'
          )
          if (course && item.course) 
            course.textContent = item.course
        }
      })
    }
  })
  .catch(() => {
    // Silently fall back to existing 
    // public/ files if API fails
  })
