    const canvas = document.getElementById('sig-canvas')
    const ctx = canvas.getContext('2d')
    ctx.strokeStyle = '#1a1a1a'
    ctx.lineWidth = 1.5
    ctx.lineCap = 'round'
    ctx.lineJoin = 'round'
    let drawing = false
    
    canvas.addEventListener('mousedown', e => {
      drawing = true
      ctx.beginPath()
      ctx.moveTo(e.offsetX, e.offsetY)
    })
    canvas.addEventListener('mousemove', e => {
      if (!drawing) return
      ctx.lineTo(e.offsetX, e.offsetY)
      ctx.stroke()
    })
    canvas.addEventListener('mouseup', () => drawing = false)
    canvas.addEventListener('mouseleave', () => drawing = false)
    
    canvas.addEventListener('touchstart', e => {
      e.preventDefault()
      drawing = true
      const r = canvas.getBoundingClientRect()
      const t = e.touches[0]
      ctx.beginPath()
      ctx.moveTo(t.clientX - r.left, t.clientY - r.top)
    }, { passive: false })
    canvas.addEventListener('touchmove', e => {
      e.preventDefault()
      if (!drawing) return
      const r = canvas.getBoundingClientRect()
      const t = e.touches[0]
      ctx.lineTo(t.clientX - r.left, t.clientY - r.top)
      ctx.stroke()
    }, { passive: false })
    canvas.addEventListener('touchend', () => drawing = false)
    
    function isCanvasBlank() {
      const pixels = ctx.getImageData(
        0, 0, canvas.width, canvas.height).data
      return !pixels.some((p, i) => i % 4 !== 3 && p !== 255)
    }

===============================================================
PART 4 — API ENDPOINTS FOR FORMS
===============================================================

POST /api/forms/medical
  Public — no auth required
  Validate bookingId exists in Supabase
  Check if medical_forms row exists for 
    this client_email + location_id:
    If exists → UPDATE the row
    If not → INSERT new row
  Capture req.ip as ip_address
  Return: { success: true, id: form_id }

POST /api/forms/consent
  Public — no auth required
  Validate bookingId exists
  Check consent not already submitted for booking
    (prevent duplicates)
  INSERT into consent_forms
  Capture req.ip as ip_address
  UPDATE bookings SET status = 'confirmed',
    forms_completed = true WHERE id = bookingId
  Send confirmation email via Resend (see Part 5)
  Return: { success: true }

GET /api/forms/check?booking=[id]
  Public
  Returns:
    booking: { id, client_name, client_email,
      treatment_name, booking_date, booking_time,
      location_id }
    medical_on_file: boolean (check by client_email)
    consent_done: boolean (check by booking_id)
  Used by /forms page on load to decide flow
