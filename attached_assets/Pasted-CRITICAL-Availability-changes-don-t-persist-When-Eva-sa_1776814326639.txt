FIX 2 — GET /api/availability/slots
MUST READ FROM SUPABASE — NO HARDCODING
===============================================================

Find the slots generation code.

Search for ANY hardcoded time strings:
'10:00', '19:00', '09:00', '16:00', '14:00', 
'08:00', '21:00', '17:00'

If ANY of these appear as fallback values 
or defaults in the slots logic → REMOVE THEM.

The slots endpoint must ONLY use Supabase:

  async function getSlots(locationId, date) {
    // Get day of week (0=Sun, 6=Sat)
    const dow = new Date(date + 'T12:00:00Z')
      .getUTCDay()
    
    // Read from Supabase
    const { data: setting } = await supabaseAdmin
      .from('availability_settings')
      .select('is_open, open_time, close_time')
      .eq('location_id', locationId)
      .eq('day_of_week', dow)
      .single()
    
    if (!setting || !setting.is_open) {
      return { available: false, 
               reason: 'CLINIC_CLOSED', slots: [] }
    }
    
    // Generate 2-hour slots from Supabase times
    const slots = []
    const [openH, openM] = setting.open_time
      .split(':').map(Number)
    const [closeH, closeM] = setting.close_time
      .split(':').map(Number)
    
    let h = openH, m = openM
    while (h * 60 + m < closeH * 60 + closeM) {
      const time = 
        String(h).padStart(2,'0') + ':' + 
        String(m).padStart(2,'0')
      slots.push({ time, available: true })
      h += 2  // 2-hour intervals
    }
    
    return { 
      available: true,
      openTime: setting.open_time,
      closeTime: setting.close_time,
      slots
    }
  }
