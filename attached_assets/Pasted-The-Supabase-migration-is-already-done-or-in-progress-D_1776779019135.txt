STEP 4 — WIRE BLOCKED SLOTS INTO BOOKING CALENDAR
===============================================================

Find the endpoint that generates available time slots
for a given date and location.

Update it to also filter out blocked_slots:

  async function getAvailableSlots(locationId, date) {
    
    const dayOfWeek = new Date(date).getDay();
    const adjustedDay = dayOfWeek === 0 ? 6 : dayOfWeek - 1;
    
    // 1. Get working hours for this day
    const { data: avail } = await supabaseAdmin
      .from('availability_settings')
      .select('open_time, close_time, is_open')
      .eq('location_id', locationId)
      .eq('day_of_week', adjustedDay)
      .single();
    
    if (!avail || !avail.is_open) return [];
    
    // 2. Generate 30-min slots between open and close
    const slots = generateTimeSlots(
      avail.open_time, avail.close_time, 30
    );
    
    // 3. Get blocked time slots for this day
    const { data: blockedSlots } = await supabaseAdmin
      .from('blocked_slots')
      .select('start_time, end_time')
      .eq('location_id', locationId)
      .or(`day_of_week.eq.${adjustedDay},all_days.eq.true`);
    
    // 4. Remove slots that fall within blocked ranges
    const filtered = slots.filter(slot => {
      return !blockedSlots?.some(b => 
        slot >= b.start_time && slot < b.end_time
      );
    });
    
    // 5. Get existing bookings for this date
    const { data: bookings } = await supabaseAdmin
      .from('bookings')
      .select('time_slot')
      .eq('location_id', locationId)
      .eq('booking_date', date)
      .in('status', ['pending', 'confirmed']);
    
    const bookedTimes = bookings?.map(b => b.time_slot) || [];
    
    // 6. Remove booked slots
    return filtered.filter(s => !bookedTimes.includes(s));
  }

  function generateTimeSlots(openTime, closeTime, intervalMins) {
    const slots = [];
    let [h, m] = openTime.split(':').map(Number);
    const [ch, cm] = closeTime.split(':').map(Number);
    const closeTotal = ch * 60 + cm;
    
    while (h * 60 + m < closeTotal) {
      slots.push(`${String(h).padStart(2,'0')}:${String(m).padStart(2,'0')}`);
      m += intervalMins;
      if (m >= 60) { h += Math.floor(m/60); m = m % 60; }
    }
    return slots;
  }
