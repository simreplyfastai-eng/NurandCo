  async function handleSignIn(email, pass) {

      method: 'POST',

      body: JSON.stringify({email, password: pass})

STEP 2 — Fix the POST endpoint.

POST /api/availability/settings must:
  Require JWT auth (X-Auth-Token header)
  Accept body: { days: [...], location: slug }
  
  For each day in the array:
    UPSERT into availability_settings:
    {
      location_id: (SELECT id FROM locations 
                    WHERE slug = req.body.location),
      day_of_week: day.day_of_week,  // 0=Sun, 6=Sat
      is_open: day.is_open,
      open_time: day.open_time,      // 'HH:MM'
      close_time: day.close_time     // 'HH:MM'
    }
    ON CONFLICT (location_id, day_of_week)
    DO UPDATE SET
      is_open = EXCLUDED.is_open,
      open_time = EXCLUDED.open_time,
      close_time = EXCLUDED.close_time
  
  Return: { success: true }

STEP 3 — Fix the GET /api/availability/slots 
endpoint to read from Supabase.

This endpoint MUST read open_time and 
close_time from Supabase availability_settings 
for the given location and day of week.

There must be ZERO hardcoded times in this 
file. Search for and delete any of:
  '10:00', '19:00', '09:00', '16:00', '14:00'
  as hardcoded string literals for slot times.

The correct logic:

  // 1. Get day of week from date param
  const date = new Date(req.query.date + 'T12:00:00Z')
  const dow = date.getUTCDay() // 0=Sun, 6=Sat
  
  // 2. Get hours from Supabase
  const { data: settings } = await supabaseAdmin
    .from('availability_settings')
    .select('is_open, open_time, close_time')
    .eq('location_id', locationId)
    .eq('day_of_week', dow)
    .single()
  
  if (!settings || !settings.is_open) {
    return res.json({ available: false, 
                      reason: 'CLINIC_CLOSED' })
  }
  
  // 3. Check blocked dates
  const { data: blocked } = await supabaseAdmin
    .from('blocked_dates')
    .select('id')
    .eq('location_id', locationId)
    .eq('date', req.query.date)
    .single()
  
  if (blocked) {
    return res.json({ available: false,
                      reason: 'DATE_BLOCKED' })
  }
  
  // 4. Generate 2-hour slots from open to close
  const slots = []
  const [openH, openM] = settings.open_time
    .split(':').map(Number)
  const [closeH, closeM] = settings.close_time
    .split(':').map(Number)
  const openMins = openH * 60 + openM
  const closeMins = closeH * 60 + closeM
  
  for (let m = openMins; m < closeMins; m += 120) {
    const h = Math.floor(m / 60)
    const min = m % 60
    const timeStr = 
      String(h).padStart(2,'0') + ':' + 
      String(min).padStart(2,'0')
    slots.push({ time: timeStr, available: true })
  }
  
  // 5. Mark booked slots as unavailable
  const { data: bookings } = await supabaseAdmin
    .from('bookings')
    .select('booking_time')
    .eq('location_id', locationId)
    .eq('booking_date', req.query.date)
    .neq('status', 'cancelled')
  
  bookings?.forEach(b => {
    const slot = slots.find(s => 
      s.time === b.booking_time?.slice(0,5))
    if (slot) slot.available = false
  })
  
  return res.json({
    available: true,
    date: req.query.date,
    dayName: date.toLocaleDateString('en-GB', 
      { weekday: 'long' }),
    openTime: settings.open_time,
    closeTime: settings.close_time,
    slots
  })
